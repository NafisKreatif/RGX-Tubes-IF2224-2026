program SquareRoot;

function AbsValue(x: real): real;
begin
  if x >= 0 then
    AbsValue := x
  else
    AbsValue := -x
end;

function SqrtApprox(x: real): real;
var
  left, right: real;
begin
  left := 0;
  right := x;
  SqrtApprox := 1.0;
  repeat
    if SqrtApprox * SqrtApprox > x then
      right := SqrtApprox
    else
      left := SqrtApprox;
    SqrtApprox := (left + right) / 2
  until AbsValue(SqrtApprox * SqrtApprox - x) < 0.0001
end;

begin
  writeln('sqrt(9) = ', SqrtApprox(9));
  writeln('sqrt(2) = ', SqrtApprox(2));
  writeln('sqrt(4/9) = ', SqrtApprox(4 / 9));
  writeln('sqrt(0.01) = ', SqrtApprox(0.01))
end.
